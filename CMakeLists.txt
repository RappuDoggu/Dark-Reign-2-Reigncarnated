cmake_minimum_required(VERSION 3.22)
project(Reigncarnated LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(REIGN_UPSTREAM "" CACHE PATH "Separate original engine checkout; never vendored")
option(REIGN_SANITIZE "Run host tests with ASan and UBSan" OFF)
option(REIGN_ANDROID_SHELL "Build an Android NativeActivity diagnostic library" OFF)
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Only arm64-v8a is validated by this harness")
endif()
add_library(reign_platform STATIC src/platform.cpp)
set_target_properties(reign_platform PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(reign_platform PUBLIC include)
if(REIGN_SANITIZE)
  if(ANDROID)
    message(FATAL_ERROR "Sanitizers here are configured for host execution only")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(reign_platform PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
endif()
if(REIGN_UPSTREAM)
  foreach(f crc.cpp crc.h random.cpp random.h packdef.h)
    if(NOT EXISTS "${REIGN_UPSTREAM}/system/${f}")
      message(FATAL_ERROR "Missing upstream system/${f}")
    endif()
  endforeach()
  include(cmake/VerifyUpstream.cmake)
  add_library(dr2_primitives STATIC "${REIGN_UPSTREAM}/system/crc.cpp" "${REIGN_UPSTREAM}/system/random.cpp")
  set_target_properties(dr2_primitives PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_options(dr2_primitives PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/include/reign/legacy_types.hpp")
  # Only CRC uses char as an unsigned lookup index. Keep this local, never global.
  set_source_files_properties("${REIGN_UPSTREAM}/system/crc.cpp" PROPERTIES COMPILE_OPTIONS "-funsigned-char")
  target_include_directories(dr2_primitives PUBLIC "${REIGN_UPSTREAM}/system" include)
  add_library(reign_checks STATIC tests/engine_checks.cpp)
  set_target_properties(reign_checks PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(reign_checks PUBLIC dr2_primitives)
  add_executable(engine_smoke tests/engine_main.cpp)
  target_link_libraries(engine_smoke PRIVATE reign_checks)
  add_test(NAME original_engine_primitives COMMAND engine_smoke)
endif()
add_executable(platform_smoke tests/platform_smoke.cpp)
target_link_libraries(platform_smoke PRIVATE reign_platform)
add_test(NAME platform_services COMMAND platform_smoke)
if(REIGN_ANDROID_SHELL)
  if(NOT ANDROID OR NOT TARGET reign_checks)
    message(FATAL_ERROR "Android diagnostic shell requires the NDK and REIGN_UPSTREAM")
  endif()
  add_library(reign_diagnostic SHARED android/native_activity.cpp)
  target_link_libraries(reign_diagnostic PRIVATE reign_checks reign_platform android log EGL GLESv3)
  target_link_options(reign_diagnostic PRIVATE -Wl,-u,ANativeActivity_onCreate)
endif()

if(NOT ANDROID)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME pack_metadata_inspector COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pack.py")
endif()
